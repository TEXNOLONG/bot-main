class VKAPIClient:
    def __init__(self, token):
        self.token = token
        self.api_version = "5.131"
        self.base_url = "https://api.vk.com/method/"
        
    def _request(self, method, params):
        """Базовый метод для запросов к VK API"""
        params.update({
            'access_token': self.token,
            'v': self.api_version
        })
        
        try:
            response = requests.get(
                self.base_url + method,
                params=params,
                verify=False,
                timeout=15
            )
            data = response.json()
            
            if 'error' in data:
                print(f"❌ Ошибка API: {data['error']['error_msg']}")
                return None
            
            return data.get('response', {})
            
        except Exception as e:
            print(f"❌ Ошибка запроса: {e}")
            return None
    
    def get_user(self, user_id, fields=None):
        """
        Получение информации о пользователе
        fields: first_name,last_name,status,sex,bdate,city,country,photo_max_orig,online,last_seen,about,activities,interests,music,movies,books,games
        """
        if fields is None:
            fields = "first_name,last_name,status,sex,bdate,city,country,photo_max_orig,online"
        
        params = {
            'user_ids': user_id,
            'fields': fields
        }
        
        result = self._request('users.get', params)
        if result and len(result) > 0:
            return result[0]
        return None
    
    def search_users(self, query, count=10, fields=None):
        """Поиск пользователей (только если токен имеет права)"""
        if fields is None:
            fields = "first_name,last_name,status,sex,photo_max_orig"
        
        params = {
            'q': query,
            'count': min(count, 100),
            'fields': fields
        }
        
        return self._request('users.search', params)
    
    def get_friends(self, user_id, count=10):
        """Получение друзей пользователя"""
        params = {
            'user_id': user_id,
            'count': min(count, 100),
            'fields': 'first_name,last_name,photo_max_orig'
        }
        
        return self._request('friends.get', params)
    
    def get_groups(self, user_id, count=10):
        """Получение групп пользователя"""
        params = {
            'user_id': user_id,
            'count': min(count, 100),
            'fields': 'name,photo_200'
        }
        
        return self._request('groups.get', params)
    
    def get_wall_posts(self, user_id, count=5):
        """Получение записей со стены пользователя"""
        params = {
            'owner_id': user_id,
            'count': min(count, 100)
        }
        
        return self._request('wall.get', params)
    
    def get_photos(self, user_id, count=5):
        """Получение фото пользователя"""
        params = {
            'owner_id': user_id,
            'count': min(count, 100),
            'album_id': 'profile'
        }
        
        return self._request('photos.get', params)

def print_user_info(user):
    """Красиво выводит информацию о пользователе"""
    if not user:
        print("❌ Пользователь не найден")
        return
    
    print("\n" + "="*60)
    print(f"👤 {user.get('first_name', '')} {user.get('last_name', '')}")
    print("="*60)
    print(f"🆔 ID: {user.get('id')}")
    print(f"♂️ Пол: {'Женский' if user.get('sex') == 1 else 'Мужской' if user.get('sex') == 2 else 'Не указан'}")
    print(f"🎂 Дата рождения: {user.get('bdate', 'Не указана')}")
    print(f"🌍 Город: {user.get('city', {}).get('title', 'Не указан')}")
    print(f"🌍 Страна: {user.get('country', {}).get('title', 'Не указана')}")
    print(f"📱 Онлайн: {'🟢 Да' if user.get('online') == 1 else '⚫ Нет'}")
    print(f"📝 Статус: {user.get('status', 'Нет')}")
    
    if user.get('last_seen'):
        import datetime
        last_seen = datetime.datetime.fromtimestamp(user['last_seen']['time'])
        print(f"⏰ Последний раз был: {last_seen.strftime('%Y-%m-%d %H:%M:%S')}")
    
    if user.get('photo_max_orig'):
        print(f"🖼️ Фото: {user['photo_max_orig']}")
    
    if user.get('about'):
        print(f"📝 О себе: {user['about']}")
